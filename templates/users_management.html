2 Aşağıdaki Hazır Kodu Projeye Yapıştırın 
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var filename = data.filename || ("isg_muratpasa_yedek_" + Utilities.formatDate(new Date(), "GMT+3", "yyyyMMdd_HHmm") + ".db");
    var decoded = Utilities.base64Decode(data.file_data);
    var blob = Utilities.newBlob(decoded, "application/x-sqlite3", filename);
    
    var folder = getBackupFolder();
    var file = folder.createFile(blob);
    
    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      fileId: file.getId(),
      fileName: file.getName()
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

// 7/24 OTOMATİK GECE YEDEĞİ (Tetikleyici ile her gece 23:00'te otomatik çeker)
function otomatikGeceYedegiAl() {
  var PORTAL_URL = "https://muratpasa-isg-portal.onrender.com/admin/veritabani-indir?token={{ backup_secret_token }}";
  try {
    var response = UrlFetchApp.fetch(PORTAL_URL, { muteHttpExceptions: true });
    if (response.getResponseCode() === 200) {
      var blob = response.getBlob();
      var filename = "isg_gece_otomatik_yedek_" + Utilities.formatDate(new Date(), "GMT+3", "yyyyMMdd_HHmm") + ".db";
      blob.setName(filename);
      var folder = getBackupFolder();
      folder.createFile(blob);
      Logger.log("Yedek alındı: " + filename);
    }
  } catch (e) {
    Logger.log("Hata: " + e.toString());
  }
}

function getBackupFolder() {
  var folderName = "Muratpasa_ISG_Yedekleri";
  var folders = DriveApp.getFoldersByName(folderName);
  return folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);
}

Google Apps Script sol menüsündeki saat simgesine (Tetikleyiciler / Triggers) tıklayın → Tetikleyici Ekle deyin → Çalıştırılacak fonksiyon olarak otomatikGeceYedegiAl seçin → Etkinlik kaynağını Zamana dayalı (Time-driven) -> Günlük zamanlayıcı -> Gece 23:00 - 00:00 yapın.
